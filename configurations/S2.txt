F2-SWITCH
=========

Enterprise Multi-Floor Network
2nd Floor Switch Configuration

enable
configure terminal

hostname F2-SWITCH

! VLAN CONFIGURATION
vlan 30
 name SALES
exit

vlan 40
 name HR
exit

vlan 50
 name FINANCE
exit

! ROUTER CONNECTION - TRUNK
interface fa0/1
 switchport mode trunk
exit

! SALES
interface range fa0/2-4
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
exit

! HR
interface range fa0/5-6
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
exit

! FINANCE
interface range fa0/7-8
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
exit

end
write memory

! Verification
show vlan brief
show interfaces trunk
